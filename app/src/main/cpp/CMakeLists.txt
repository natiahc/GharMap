cmake_minimum_required(VERSION 3.22)
project(gharmap_android LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_subdirectory("${ROOT}/core" "${CMAKE_CURRENT_BINARY_DIR}/core")
add_library(arcore SHARED IMPORTED)
set_target_properties(arcore PROPERTIES IMPORTED_LOCATION "${ARCORE_LIB_DIR}/${ANDROID_ABI}/libarcore_sdk_c.so")
add_library(gharmap SHARED native_bridge.cpp)
target_include_directories(gharmap PRIVATE "${ROOT}/third_party/arcore/include")
target_compile_features(gharmap PRIVATE cxx_std_17)
target_link_libraries(gharmap PRIVATE gharmap_core arcore GLESv2 log android)
