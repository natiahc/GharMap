cmake_minimum_required(VERSION 3.22)
project(gharmap_core LANGUAGES CXX)
add_library(gharmap_core STATIC src/geometry.cpp)
target_include_directories(gharmap_core PUBLIC include)
target_compile_features(gharmap_core PUBLIC cxx_std_17)
set_target_properties(gharmap_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID)
  enable_testing()
  add_executable(core_tests tests/geometry_test.cpp)
  target_link_libraries(core_tests PRIVATE gharmap_core)
  add_test(NAME geometry COMMAND core_tests)
endif()
