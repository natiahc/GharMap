cmake_minimum_required(VERSION 3.22)
project(paisapilot_finance LANGUAGES CXX)
add_library(paisapilot_finance STATIC src/finance.cpp)
target_include_directories(paisapilot_finance PUBLIC include)
target_compile_features(paisapilot_finance PUBLIC cxx_std_17)
set_target_properties(paisapilot_finance PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID AND NOT IOS)
  enable_testing()
  add_executable(finance_tests tests/finance_test.cpp)
  target_link_libraries(finance_tests PRIVATE paisapilot_finance)
  add_test(NAME finance COMMAND finance_tests)
endif()
